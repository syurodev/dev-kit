	gitservice "devkit/internal/git"

func probeGit(t *testing.T, dir string, args ...string) {
	t.Helper()
	cmd := exec.Command("git", append([]string{"-C", dir}, args...)...)
	cmd.Env = append(os.Environ(), "GIT_AUTHOR_NAME=t", "GIT_AUTHOR_EMAIL=t@t", "GIT_COMMITTER_NAME=t", "GIT_COMMITTER_EMAIL=t@t")
	if out, err := cmd.CombinedOutput(); err != nil {
		t.Fatalf("git %v: %v: %s", args, err, out)
	}
}

func probeRepo(t *testing.T, dir, subject string) {
	t.Helper()
	if err := os.MkdirAll(dir, 0o700); err != nil {
		t.Fatal(err)
	}
	probeGit(t, dir, "init", "-q")
	probeGit(t, dir, "commit", "-q", "--allow-empty", "-m", subject)
}

// R1a probe, deterministic half: after gitFor has resolved, checked and opened a
// real directory, replacing that directory with a link to another repository
// makes the next command of the SAME returned handle run in the other repository.
func TestR1aProbeDirectorySwapAfterOpenRunsInAnotherRepository(t *testing.T) {
	svc := gitservice.NewService(context.Background())
	if !svc.Available() {
		t.Skip("no git")
	}
	srv, root, outside := linkedScopeServer(t, GitReposOf(svc))
	pkg := filepath.Join(root, "pkg")
	probeRepo(t, pkg, "GRANTED-REPO")
	probeRepo(t, outside, "VICTIM-REPO-OUTSIDE-THE-GRANT")

	api, _, err := srv.gitFor(agentCtx(), pkg, false)
	if err != nil {
		t.Fatal(err)
	}
	before, err := api.Log(context.Background(), 1)
	if err != nil || len(before) != 1 {
		t.Fatalf("log before: %v %v", before, err)
	}
	// The swap an agent with write access to the granted folder can make.
	if err := os.Rename(pkg, pkg+".moved"); err != nil {
		t.Fatal(err)
	}
	if err := os.Symlink(outside, pkg); err != nil {
		t.Fatal(err)
	}
	after, err := api.Log(context.Background(), 1)
	if err != nil || len(after) != 1 {
		t.Fatalf("log after: %v %v", after, err)
	}
	t.Logf("same handle: before swap %q, after swap %q", before[0].Subject, after[0].Subject)
	if strings.Contains(after[0].Subject, "VICTIM") {
		t.Errorf("DIRECTORY SWAP AFTER OPEN: the handle checked for %s now runs git in a repository outside the grant", pkg)
	}
}

// R1a probe, statistical half: a flipper that atomically exchanges the real directory
// and a link to the other repository (renameat2 RENAME_EXCHANGE) while the tool path
// is called in a loop. Counts how many complete calls returned the other repository.
func TestR1aProbeDirectorySwapRaceIsWinnable(t *testing.T) {
	svc := gitservice.NewService(context.Background())
	if !svc.Available() {
		t.Skip("no git")
	}
	srv, root, outside := linkedScopeServer(t, GitReposOf(svc))
	pkg, other := filepath.Join(root, "pkg"), filepath.Join(root, "pkg.x")
	probeRepo(t, pkg, "GRANTED-REPO")
	probeRepo(t, outside, "VICTIM-REPO-OUTSIDE-THE-GRANT")
	if err := os.Symlink(outside, other); err != nil {
		t.Fatal(err)
	}
	var stop atomic.Bool
	done := make(chan struct{})
	var flips atomic.Int64
	go func() {
		defer close(done)
		for !stop.Load() {
			if err := unix.Renameat2(unix.AT_FDCWD, pkg, unix.AT_FDCWD, other, unix.RENAME_EXCHANGE); err != nil {
				t.Errorf("exchange: %v", err)
				return
			}
			flips.Add(1)
		}
	}()
	granted, victim, refused, other_ := 0, 0, 0, 0
	const iterations = 400
	for i := 0; i < iterations; i++ {
		api, _, err := srv.gitFor(agentCtx(), pkg, false)
		if err != nil {
			refused++
			continue
		}
		commits, err := api.Log(context.Background(), 1)
		switch {
		case err != nil || len(commits) != 1:
			other_++
		case strings.Contains(commits[0].Subject, "VICTIM"):
			victim++
		default:
			granted++
		}
	}
	stop.Store(true)
	<-done
	t.Logf("iterations=%d flips=%d | refused at open=%d, answered from the granted repo=%d, answered from the OUTSIDE repo=%d, other=%d", iterations, flips.Load(), refused, granted, victim, other_)
	if victim > 0 {
		t.Errorf("RACE WON: %d of %d calls were answered from a repository outside the grant", victim, iterations)
	}
}

// R1a probe (adjacent, not the race): repository-controlled configuration is run by the
// host's git. Does a granted repository's own `core.fsmonitor` execute when the git
// tool reads status? (A harmless `touch` of a marker file.)
func TestR1aProbeRepositoryConfigRunsInTheHostsGit(t *testing.T) {
	svc := gitservice.NewService(context.Background())
	if !svc.Available() {
		t.Skip("no git")
	}
	srv, root, _ := linkedScopeServer(t, GitReposOf(svc))
	pkg := filepath.Join(root, "pkg")
	probeRepo(t, pkg, "GRANTED-REPO")
	marker := filepath.Join(t.TempDir(), "ran")
	probeGit(t, pkg, "config", "core.fsmonitor", "touch "+marker+"; true")
	api, _, err := srv.gitFor(agentCtx(), pkg, false)
	if err != nil {
		t.Fatal(err)
	}
	_, _ = api.Status(context.Background())
	if _, err := os.Stat(marker); err == nil {
		t.Errorf("HOST EXEC: the repository's core.fsmonitor command ran in the host's git process")
	}
}

// R1a probe, no race: a directory inside the granted folder whose `.git` is a FILE
// (`gitdir: <other repository>/.git`) is a work tree of the other repository to Git.
// The Toplevel check passes (the work tree root is the granted directory).
func TestR1aProbeGitdirFileRedirectsAGrantedFolderToAnotherRepository(t *testing.T) {
	svc := gitservice.NewService(context.Background())
	if !svc.Available() {
		t.Skip("no git")
	}
	srv, root, outside := linkedScopeServer(t, GitReposOf(svc))
	probeRepo(t, outside, "VICTIM-REPO-OUTSIDE-THE-GRANT")
	pkg := filepath.Join(root, "pkg")
	if err := os.WriteFile(filepath.Join(pkg, ".git"), []byte("gitdir: "+filepath.Join(outside, ".git")+"\n"), 0o600); err != nil {
		t.Fatal(err)
	}
	api, _, err := srv.gitFor(agentCtx(), pkg, false)
	if err != nil {
		t.Logf("refused at open: %v", err)
		return
	}
	commits, err := api.Log(context.Background(), 1)
	if err != nil || len(commits) != 1 {
		t.Fatalf("log: %v %v", commits, err)
	}
	t.Logf("handle for %s answers from: %q", pkg, commits[0].Subject)
	if strings.Contains(commits[0].Subject, "VICTIM") {
		t.Errorf("GITDIR FILE: a granted folder redirected to a repository outside the grant, no race needed")
	}
}
